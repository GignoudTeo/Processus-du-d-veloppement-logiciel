
BEGIN;

-- ========================================================
-- NETTOYAGE DES TABLES EXISTANTES
-- ========================================================

DROP TABLE IF EXISTS reservations CASCADE;
DROP TABLE IF EXISTS cours CASCADE;
DROP TABLE IF EXISTS sports CASCADE;
DROP TABLE IF EXISTS categories CASCADE;
DROP TABLE IF EXISTS salles CASCADE;
DROP TABLE IF EXISTS admins CASCADE;
DROP TABLE IF EXISTS coachs CASCADE;
DROP TABLE IF EXISTS adherents CASCADE;
DROP TABLE IF EXISTS comptes CASCADE;

DROP TYPE IF EXISTS role_compte CASCADE;
DROP TYPE IF EXISTS statut_reservation CASCADE;

-- ========================================================
-- 0. TYPES ENUM
-- ========================================================

CREATE TYPE role_compte AS ENUM ('adherent', 'coach', 'admin');

CREATE TYPE statut_reservation AS ENUM (
    'confirmee',
    'annulee',
    'liste_attente'
);

-- ========================================================
-- 1. COMPTES
-- ========================================================

CREATE TABLE comptes (
    id_compte INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    email VARCHAR(191) NOT NULL UNIQUE,
    mot_de_passe_hash VARCHAR(255) NOT NULL,
    role role_compte NOT NULL DEFAULT 'adherent',
    date_creation TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP,
    email_confirme BOOLEAN NOT NULL DEFAULT FALSE,
    jeton_confirmation CHAR(64) DEFAULT NULL
);

-- ========================================================
-- 2. ADHERENTS, COACHS ET ADMINISTRATEURS
-- Structure originale conservee
-- ========================================================

CREATE TABLE adherents (
    id_adherent INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    id_compte INTEGER NOT NULL UNIQUE
        REFERENCES comptes(id_compte)
        ON DELETE CASCADE ON UPDATE CASCADE,
    nom VARCHAR(100) NOT NULL,
    prenom VARCHAR(100) NOT NULL,
    adresse VARCHAR(255) DEFAULT NULL,
    poids DECIMAL(5,2) DEFAULT NULL,
    taille DECIMAL(5,2) DEFAULT NULL,
    telephone VARCHAR(20) DEFAULT NULL,
    date_naissance DATE DEFAULT NULL
);

COMMENT ON COLUMN adherents.poids IS 'Poids en kg';
COMMENT ON COLUMN adherents.taille IS 'Taille en cm';

CREATE TABLE coachs (
    id_coach INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    id_compte INTEGER NOT NULL UNIQUE
        REFERENCES comptes(id_compte)
        ON DELETE CASCADE ON UPDATE CASCADE,
    nom VARCHAR(100) NOT NULL,
    prenom VARCHAR(100) NOT NULL,
    specialite VARCHAR(150) DEFAULT NULL,
    bio TEXT DEFAULT NULL
);

CREATE TABLE admins (
    id_admin INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    id_compte INTEGER NOT NULL UNIQUE
        REFERENCES comptes(id_compte)
        ON DELETE CASCADE ON UPDATE CASCADE,
    nom VARCHAR(100) NOT NULL,
    prenom VARCHAR(100) NOT NULL,
    niveau_acces VARCHAR(50) NOT NULL DEFAULT 'standard'
);

-- ========================================================
-- 3. CATEGORIES, SPORTS ET SALLES
-- ========================================================

CREATE TABLE categories (
    id_categorie INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    nom_categorie VARCHAR(100) NOT NULL UNIQUE
);

CREATE TABLE sports (
    id_sport INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    id_categorie INTEGER NOT NULL
        REFERENCES categories(id_categorie)
        ON DELETE RESTRICT ON UPDATE CASCADE,
    nom_sport VARCHAR(100) NOT NULL,
    description TEXT DEFAULT NULL
);

CREATE INDEX idx_sports_categorie ON sports(id_categorie);

CREATE TABLE salles (
    id_salle INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    nom_salle VARCHAR(100) NOT NULL,
    capacite SMALLINT NOT NULL,
    adresse VARCHAR(255) DEFAULT NULL
);

-- ========================================================
-- 4. COURS : STRUCTURE ORIGINALE CONSERVEE
-- ========================================================

CREATE TABLE cours (
    id_cours INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    titre VARCHAR(150) NOT NULL,

    id_sport INTEGER NOT NULL
        REFERENCES sports(id_sport)
        ON DELETE RESTRICT ON UPDATE CASCADE,

    id_coach INTEGER NOT NULL
        REFERENCES coachs(id_coach)
        ON DELETE RESTRICT ON UPDATE CASCADE,

    id_salle INTEGER NOT NULL
        REFERENCES salles(id_salle)
        ON DELETE RESTRICT ON UPDATE CASCADE,

    date_debut TIMESTAMPTZ NOT NULL,
    date_fin TIMESTAMPTZ NOT NULL,
    nb_places_max SMALLINT NOT NULL,

    CONSTRAINT chk_cours_dates
        CHECK (date_fin > date_debut),

    CONSTRAINT chk_cours_places
        CHECK (nb_places_max > 0)
);

CREATE INDEX idx_cours_sport ON cours(id_sport);
CREATE INDEX idx_cours_coach ON cours(id_coach);
CREATE INDEX idx_cours_salle ON cours(id_salle);

-- ========================================================
-- 5. RESERVATIONS
-- Structure originale conservee
-- ========================================================

CREATE TABLE reservations (
    id_reservation INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,

    id_cours INTEGER NOT NULL
        REFERENCES cours(id_cours)
        ON DELETE CASCADE ON UPDATE CASCADE,

    id_adherent INTEGER NOT NULL
        REFERENCES adherents(id_adherent)
        ON DELETE CASCADE ON UPDATE CASCADE,

    date_reservation TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP,
    statut statut_reservation NOT NULL DEFAULT 'confirmee',

    UNIQUE (id_cours, id_adherent)
);

CREATE INDEX idx_reservations_adherent
    ON reservations(id_adherent);

-- ========================================================
-- 12. PLANIFICATION CORRIGEE DES COURS
-- ========================================================
-- Les sports sont repartis sur des dates differentes.
-- Les cours de chaque coach sont programmes dans les
-- cinq salles avec des horaires distincts.
--
-- Salle 1 : 08 h - 09 h
-- Salle 2 : 10 h - 11 h
-- Salle 3 : 12 h - 13 h
-- Salle 4 : 14 h - 15 h
-- Salle 5 : 16 h - 17 h
--
-- Les heures sont exprimees en heure locale de Paris.
-- Les combinaisons salle/sport deja existantes sont ignorees.

WITH correspondance_sport_coach (id_sport, id_coach) AS (
    VALUES
        (1, 2), -- HIIT : Thomas Moreau
        (2, 3), -- Body Pump : Julie Lefevre
        (3, 4), -- Yoga : Sophie Petit
        (4, 5), -- Pilates : Emma Garcia
        (5, 6), -- Cycling : Antoine Roux
        (6, 7), -- Boxe Fitness : Lucas Fournier
        (7, 8)  -- Zumba : Camille Girard
),
combinaisons AS (
    SELECT
        sa.id_salle,
        sa.nom_salle,
        sa.capacite,
        sp.id_sport,
        sp.nom_sport,
        sc.id_coach
    FROM salles sa
    CROSS JOIN sports sp
    JOIN correspondance_sport_coach sc
        ON sc.id_sport = sp.id_sport
    WHERE NOT EXISTS (
        SELECT 1
        FROM cours c
        WHERE c.id_salle = sa.id_salle
          AND c.id_sport = sp.id_sport
    )
)
INSERT INTO cours (
    titre,
    id_sport,
    id_coach,
    id_salle,
    date_debut,
    date_fin,
    nb_places_max
)
SELECT
    CASE id_sport
        WHEN 1 THEN 'HIIT - Cardio explosif'
        WHEN 2 THEN 'Body Pump - Renforcement complet'
        WHEN 3 THEN 'Yoga - Souplesse et respiration'
        WHEN 4 THEN 'Pilates - Posture et muscles profonds'
        WHEN 5 THEN 'Cycling - Endurance cardio'
        WHEN 6 THEN 'Boxe Fitness - Technique et cardio'
        WHEN 7 THEN 'Zumba - Danse et énergie'
    END || ' - ' || nom_salle AS titre,

    id_sport,
    id_coach,
    id_salle,

    (
        (
            DATE '2026-11-01' + (id_sport - 1)
        )::timestamp
        + TIME '08:00'
        + ((id_salle - 1) * INTERVAL '2 hours')
    ) AT TIME ZONE 'Europe/Paris' AS date_debut,

    (
        (
            DATE '2026-11-01' + (id_sport - 1)
        )::timestamp
        + TIME '09:00'
        + ((id_salle - 1) * INTERVAL '2 hours')
    ) AT TIME ZONE 'Europe/Paris' AS date_fin,

    LEAST(capacite, 15) AS nb_places_max

FROM combinaisons;
