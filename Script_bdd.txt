CREATE TYPE role_compte AS ENUM ('adherent', 'coach', 'admin');

CREATE TYPE statut_reservation AS ENUM (
    'en_attente',
    'confirmee',
    'annulee',
    'liste_attente',
    'rejetee',
    'absent'
);

-- 1. COMPTES
-- ========================================================

    id_compte INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,

    role role_compte NOT NULL DEFAULT 'adherent',

    email_confirme BOOLEAN NOT NULL DEFAULT FALSE,

-- 2. ADHERENTS, COACHS ET ADMINISTRATEURS
-- Structure originale conservee

    taille DECIMAL(5,2) DEFAULT NULL,
    telephone VARCHAR(20) DEFAULT NULL,
    date_naissance DATE DEFAULT NULL,

    -- Statistiques des reservations de l'adherent
    nb_reservations INTEGER NOT NULL DEFAULT 0
        CHECK (nb_reservations >= 0),
    nb_reservations_rejetees INTEGER NOT NULL DEFAULT 0
        CHECK (nb_reservations_rejetees >= 0),
    nb_absences INTEGER NOT NULL DEFAULT 0
        CHECK (nb_absences >= 0)

-- 4. COURS : STRUCTURE ORIGINALE CONSERVEE

CREATE TABLE cours (
    id_cours INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    titre VARCHAR(150) NOT NULL,
    description TEXT DEFAULT NULL,

    id_sport INTEGER NOT NULL
        REFERENCES sports(id_sport)
        ON DELETE RESTRICT ON UPDATE CASCADE,

    id_coach INTEGER NOT NULL
        REFERENCES coachs(id_coach)
        ON DELETE RESTRICT ON UPDATE CASCADE,

    id_salle INTEGER NOT NULL
        REFERENCES salles(id_salle)
        ON DELETE RESTRICT ON UPDATE CASCADE,

    date_debut TIMESTAMPTZ NOT NULL,
    date_fin TIMESTAMPTZ NOT NULL,
    nb_places_max SMALLINT NOT NULL,

    -- Rappel J-3 au coach : envoyé une seule fois par séance
    rappel_envoye_le TIMESTAMPTZ DEFAULT NULL,

    CONSTRAINT chk_cours_dates
        CHECK (date_fin > date_debut),

    CONSTRAINT chk_cours_places
        CHECK (nb_places_max > 0)
);

CREATE INDEX idx_cours_sport ON cours(id_sport);
CREATE INDEX idx_cours_coach ON cours(id_coach);
CREATE INDEX idx_cours_salle ON cours(id_salle);

-- ========================================================
-- 5. RESERVATIONS
-- Structure originale conservee
-- ========================================================

CREATE TABLE reservations (
    id_reservation INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,

    id_cours INTEGER NOT NULL
        REFERENCES cours(id_cours)
        ON DELETE CASCADE ON UPDATE CASCADE,

    id_adherent INTEGER NOT NULL
        REFERENCES adherents(id_adherent)
        ON DELETE CASCADE ON UPDATE CASCADE,

    date_reservation TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP,
    statut statut_reservation NOT NULL DEFAULT 'en_attente',

    -- Motif et date du dernier changement de statut (refus, désinscription...)
    motif TEXT DEFAULT NULL,
    date_statut TIMESTAMPTZ DEFAULT NULL,

    UNIQUE (id_cours, id_adherent)
);

CREATE INDEX idx_reservations_adherent
    ON reservations(id_adherent);

-- ========================================================
-- MISE A JOUR AUTOMATIQUE DES STATISTIQUES DES ADHERENTS
-- ========================================================
-- nb_reservations : nombre total de reservations enregistrees
-- nb_reservations_rejetees : reservations avec le statut 'rejetee'
-- nb_absences : reservations avec le statut 'absent'
-- Les compteurs sont recalcules apres chaque ajout, modification
-- ou suppression d'une reservation.

CREATE OR REPLACE FUNCTION actualiser_statistiques_adherent()
RETURNS TRIGGER AS $$
BEGIN
    -- Recalcul de l'ancien adherent en cas de modification
    -- ou de suppression d'une reservation.
    IF TG_OP = 'UPDATE' OR TG_OP = 'DELETE' THEN
        UPDATE adherents
        SET
            nb_reservations = (
                SELECT COUNT(*)::INTEGER
                FROM reservations
                WHERE id_adherent = OLD.id_adherent
            ),
            nb_reservations_rejetees = (
                SELECT COUNT(*)::INTEGER
                FROM reservations
                WHERE id_adherent = OLD.id_adherent
                  AND statut = 'rejetee'
            ),
            nb_absences = (
                SELECT COUNT(*)::INTEGER
                FROM reservations
                WHERE id_adherent = OLD.id_adherent
                  AND statut = 'absent'
            )
        WHERE id_adherent = OLD.id_adherent;
    END IF;

    -- Recalcul du nouvel adherent en cas d'ajout
    -- ou de modification d'une reservation.
    IF TG_OP = 'INSERT' OR TG_OP = 'UPDATE' THEN
        UPDATE adherents
        SET
            nb_reservations = (
                SELECT COUNT(*)::INTEGER
                FROM reservations
                WHERE id_adherent = NEW.id_adherent
            ),
            nb_reservations_rejetees = (
                SELECT COUNT(*)::INTEGER
                FROM reservations
                WHERE id_adherent = NEW.id_adherent
                  AND statut = 'rejetee'
            ),
            nb_absences = (
                SELECT COUNT(*)::INTEGER
                FROM reservations
                WHERE id_adherent = NEW.id_adherent
                  AND statut = 'absent'
            )
        WHERE id_adherent = NEW.id_adherent;
    END IF;

    RETURN NULL;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trigger_actualiser_statistiques_adherent
AFTER INSERT OR UPDATE OR DELETE
ON reservations
FOR EACH ROW
EXECUTE FUNCTION actualiser_statistiques_adherent();


-- ========================================================
-- 5 ter. HISTORIQUE, PASSAGES, PROGRESSION ET INCIDENTS
-- ========================================================

-- Refus, désinscriptions et annulations de séance.
-- Les noms sont recopiés : l'historique reste lisible même si la séance,
-- le coach ou l'adhérent est supprimé ensuite (pas de clé étrangère sur id_cours).
CREATE TABLE IF NOT EXISTS historique_seances (
    id_evenement INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    type VARCHAR(20) NOT NULL
        CHECK (type IN ('refus', 'desinscription', 'annulation')),
    date_evenement TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP,
    motif TEXT NOT NULL,

    id_cours INTEGER NOT NULL,
    titre_cours VARCHAR(150) NOT NULL,
    date_cours TIMESTAMPTZ NOT NULL,
    nom_salle VARCHAR(100) DEFAULT NULL,

    id_coach INTEGER DEFAULT NULL
        REFERENCES coachs(id_coach)
        ON DELETE SET NULL ON UPDATE CASCADE,
    nom_coach VARCHAR(201) DEFAULT NULL,

    -- NULL pour la ligne "séance annulée" vue côté coach
    id_adherent INTEGER DEFAULT NULL
        REFERENCES adherents(id_adherent)
        ON DELETE SET NULL ON UPDATE CASCADE,
    nom_adherent VARCHAR(201) DEFAULT NULL
);

CREATE INDEX IF NOT EXISTS idx_historique_type_date ON historique_seances(type, date_evenement);
CREATE INDEX IF NOT EXISTS idx_historique_coach ON historique_seances(id_coach);
CREATE INDEX IF NOT EXISTS idx_historique_adherent ON historique_seances(id_adherent);

-- Entrées et sorties de la salle (saisies à la main, pas de lecteur de badge)
CREATE TABLE IF NOT EXISTS passages (
    id_passage INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    id_adherent INTEGER NOT NULL
        REFERENCES adherents(id_adherent)
        ON DELETE CASCADE ON UPDATE CASCADE,
    entree TIMESTAMPTZ NOT NULL,
    -- NULL : passage incomplet (sortie non enregistrée)
    sortie TIMESTAMPTZ DEFAULT NULL,
    CONSTRAINT chk_passages_sortie CHECK (sortie IS NULL OR sortie > entree)
);

CREATE INDEX IF NOT EXISTS idx_passages_adherent_entree ON passages(id_adherent, entree);

-- Mesures saisies par l'adhérent. L'IMC est calculé par le site, il n'est pas stocké.
CREATE TABLE IF NOT EXISTS progression (
    id_mesure INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    id_adherent INTEGER NOT NULL
        REFERENCES adherents(id_adherent)
        ON DELETE CASCADE ON UPDATE CASCADE,
    date_mesure DATE NOT NULL,
    poids DECIMAL(5,2) NOT NULL CHECK (poids BETWEEN 20 AND 300),
    taille DECIMAL(5,2) NOT NULL CHECK (taille BETWEEN 100 AND 250),
    duree_seance SMALLINT NOT NULL CHECK (duree_seance BETWEEN 1 AND 600)
);

CREATE INDEX IF NOT EXISTS idx_progression_adherent_date ON progression(id_adherent, date_mesure);

-- Incidents signalés par un coach sur une de ses séances
CREATE TABLE IF NOT EXISTS incidents (
    id_incident INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    id_coach INTEGER NOT NULL
        REFERENCES coachs(id_coach)
        ON DELETE CASCADE ON UPDATE CASCADE,
    id_cours INTEGER NOT NULL,
    titre_cours VARCHAR(150) NOT NULL,
    date_cours TIMESTAMPTZ NOT NULL,
    date_signalement TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP,
    description TEXT NOT NULL
);

CREATE INDEX IF NOT EXISTS idx_incidents_coach ON incidents(id_coach);

-- ========================================================
-- 12. PLANIFICATION CORRIGEE DES COURS
-- ========================================================
-- Les sports sont repartis sur des dates differentes.
-- Les cours de chaque coach sont programmes dans les
-- cinq salles avec des horaires distincts.
--
-- Salle 1 : 08 h - 09 h
-- Salle 2 : 10 h - 11 h
-- Salle 3 : 12 h - 13 h
-- Salle 4 : 14 h - 15 h
-- Salle 5 : 16 h - 17 h
--
-- Les heures sont exprimees en heure locale de Paris.
-- Les combinaisons salle/sport deja existantes sont ignorees.

WITH correspondance_sport_coach (id_sport, id_coach) AS (
    VALUES
        (1, 2), -- HIIT : Thomas Moreau
        (2, 3), -- Body Pump : Julie Lefevre
        (3, 4), -- Yoga : Sophie Petit
        (4, 5), -- Pilates : Emma Garcia
        (5, 6), -- Cycling : Antoine Roux
        (6, 7), -- Boxe Fitness : Lucas Fournier
        (7, 8)  -- Zumba : Camille Girard
),
combinaisons AS (
    SELECT
        sa.id_salle,
        sa.nom_salle,
        sa.capacite,
        sp.id_sport,
        sp.nom_sport,
        sc.id_coach
    FROM salles sa
    CROSS JOIN sports sp
    JOIN correspondance_sport_coach sc
        ON sc.id_sport = sp.id_sport
    WHERE NOT EXISTS (
        SELECT 1
        FROM cours c
        WHERE c.id_salle = sa.id_salle
          AND c.id_sport = sp.id_sport
    )
)
INSERT INTO cours (
    titre,
    id_sport,
    id_coach,
    id_salle,
    date_debut,
    date_fin,
    nb_places_max
)
SELECT
    CASE id_sport
        WHEN 1 THEN 'HIIT - Cardio explosif'
        WHEN 2 THEN 'Body Pump - Renforcement complet'
        WHEN 3 THEN 'Yoga - Souplesse et respiration'
        WHEN 4 THEN 'Pilates - Posture et muscles profonds'
        WHEN 5 THEN 'Cycling - Endurance cardio'
        WHEN 6 THEN 'Boxe Fitness - Technique et cardio'
        WHEN 7 THEN 'Zumba - Danse et énergie'
    END || ' - ' || nom_salle AS titre,

    id_sport,
    id_coach,
    id_salle,

    (
        (
            DATE '2026-11-01' + (id_sport - 1)
        )::timestamp
        + TIME '08:00'
        + ((id_salle - 1) * INTERVAL '2 hours')
    ) AT TIME ZONE 'Europe/Paris' AS date_debut,

    (
        (
            DATE '2026-11-01' + (id_sport - 1)
        )::timestamp
        + TIME '09:00'
        + ((id_salle - 1) * INTERVAL '2 hours')
    ) AT TIME ZONE 'Europe/Paris' AS date_fin,

    LEAST(capacite, 15) AS nb_places_max

FROM combinaisons;

-- ========================================================
-- JEU DE DONNÉES : passages en salle
-- ========================================================
-- Les 5 premiers adhérents, sur les 6 dernières semaines : 3 venues par semaine
-- (lundi, mercredi, samedi), de 45 min à 1 h 45. Quelques passages sans sortie
-- pour tester l'affichage "incomplet".

INSERT INTO passages (id_adherent, entree, sortie)
SELECT
    a.id_adherent,
    jour + (TIME '17:00' + (a.rang * 30) * INTERVAL '1 minute'),
    CASE
        WHEN (a.rang + semaine) % 4 = 0 AND extract(isodow FROM jour) = 6 THEN NULL
        ELSE jour + (TIME '17:00' + (a.rang * 30) * INTERVAL '1 minute')
                  + ((45 + ((a.rang * 15 + semaine * 10) % 60)) * INTERVAL '1 minute')
    END
FROM (
    SELECT id_adherent, ROW_NUMBER() OVER (ORDER BY id_adherent) - 1 AS rang
    FROM adherents
    ORDER BY id_adherent
    LIMIT 5
) a
CROSS JOIN generate_series(1, 6) AS semaine
CROSS JOIN LATERAL (
    SELECT (date_trunc('week', CURRENT_DATE) - (semaine * INTERVAL '1 week') + d * INTERVAL '1 day')::date AS jour
    FROM unnest(ARRAY[0, 2, 5]) AS d
) jours;
